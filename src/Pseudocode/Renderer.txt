// Unified Renderer - All rendering in one place
// Handles planets, trails, labels, UI, and grid

Define class Renderer
    constructor(canvasManager, coordSystem)
        // Get canvas and drawing context from manager
        Assign canvasManager.canvas to this.canvas
        Assign canvasManager.ctx to this.ctx
        
        // Store coordinate system for converting positions
        Assign coordSystem to this.coordSystem
        
        // Cache for loaded planet sprites (images)
        Assign empty object to this.spriteCache
        
        // Set image smoothing false for better quality
        Set this.ctx.imageSmoothingEnabled to false

    // MARK: CLEAR SCREEN
    clear(bgColor)
        // Fill entire canvas with background color
        Assign bgColor to this.ctx.fillStyle
        Call this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height)

    // MARK: PLANET RENDERING
    drawPlanet(body, cameraMode, useRealScale, minSize, scaleBoost)
        // Skip drawing barycenters except in focus mode (mode 3)
        If body.isBarycenter AND cameraMode is not equal to 3
            Return
        
        // Convert world position to screen coordinates
        Assign this.coordSystem.worldToScreen(body.position.x, body.position.y) to screenPos
        
        // If it's a barycenter, just draw a small marker
        If body.isBarycenter
            Call this.drawBarycenterMarker(screenPos)
            Return
        
        // Calculate how big to draw the planet
        Assign this.calculateRadius(body, cameraMode, useRealScale, minSize, scaleBoost) to radius
        
        // Draw planet sprite if it has one
        If body.spritePath exists
            // Don't draw Moon sprite in modes 1 and 2 (too small)
            If NOT (body.name equals "Moon" AND (cameraMode equals 1 OR cameraMode equals 2))
                Call this.drawPlanetSprite(body, screenPos, radius)
        
        // Draw label if planet is big enough
        If radius is greater than or equal to 2 AND NOT (body.name equals "Earth-Moon Barycenter")
            // Don't draw Moon label in modes 1 and 2
            If NOT (body.name equals "Moon" AND (cameraMode equals 1 OR cameraMode equals 2))
                Call this.drawPlanetLabel(body.name, screenPos.x, screenPos.y - radius - 8, radius, cameraMode)

    drawPlanetSprite(body, screenPos, radius)
        // Get sprite path as cache key
        Assign body.spritePath to key
        
        // If sprite already loaded, draw it
        If this.spriteCache[key] exists
            Call this.ctx.drawImage(this.spriteCache[key], screenPos.x - radius, screenPos.y - radius, radius * 2, radius * 2)
            Return
        
        // Otherwise, load the sprite
        Create new Image object, assign to img
        Set img.onload function to:
            Store img in this.spriteCache[key]
        Set img.src to key

    drawBarycenterMarker(screenPos)
        // Draw a small white dot for barycenter
        Assign 'rgba(255, 255, 255, 0.3)' to this.ctx.fillStyle
        Call this.ctx.beginPath()
        Call this.ctx.arc(screenPos.x, screenPos.y, 2, 0, Math.PI * 2)
        Call this.ctx.fill()

    calculateRadius(body, cameraMode, useRealScale, minSize, scaleBoost)
        // If using artistic/logarithmic scaling
        If useRealScale is false
            // Scale based on mass (logarithmic)
            Assign body.mass divided by 1e20 to temp
            Assign Math.log10(temp) to logValue
            Assign logValue multiplied by 2 to base
            Assign Math.max(2, base) to base
            
            // In focus mode, apply scale boost
            If cameraMode equals 3
                Return base multiplied by scaleBoost
            Otherwise
                Return base
        
        // If using real-world scaling
        // Convert radius from km to meters
        Assign body.radius multiplied by 1000 to worldRadius
        
        // Get scaling multipliers from coordinate system
        Assign this.coordSystem.getPlanetSizeMultipliers() to multipliers
        
        // Sun gets different multiplier than planets
        If body.name equals 'Sun'
            Assign multipliers.sunMultiplier to multiplier
        Otherwise
            Assign multipliers.planetMultiplier to multiplier
        
        // Calculate screen radius
        Assign (worldRadius divided by this.coordSystem.metersPerPixel) multiplied by multiplier to screenRadius
        
        // Set minimum size based on camera mode
        Assign minSize to minForMode
        If cameraMode equals 3
            Assign Math.max(8, minSize) to minForMode
        Otherwise if cameraMode equals 2
            Assign Math.max(4, minSize) to minForMode
        Otherwise
            Assign Math.max(3, minSize) to minForMode
        
        // Return larger of minimum and calculated radius
        Return Math.max(minForMode, screenRadius)

    drawPlanetLabel(text, x, y, radius, cameraMode)
        // Default font size
        Assign 12 to fontSize
        
        // Adjust font size based on camera mode and planet size
        If cameraMode equals 3
            Assign Math.min(16, Math.max(10, radius multiplied by 0.6)) to fontSize
        Otherwise if cameraMode equals 2
            Assign Math.min(14, Math.max(8, radius multiplied by 0.8)) to fontSize
        
        // Draw the label
        Assign 'white' to this.ctx.fillStyle
        Assign fontSize concatenated with 'px Arial' to this.ctx.font
        Assign 'center' to this.ctx.textAlign
        Call this.ctx.fillText(text, x, y)

    // MARK: TRAIL RENDERING
    drawTrail(body)
        // Don't draw trails for Moon or satellites
        If NOT body exists OR body.name equals 'Moon' OR String(body.satellite) equals 'true'
            Return
        
        // Need at least 2 points to draw a trail
        If NOT body.trail exists OR body.trail.length is less than 2
            Return
        
        // Convert all trail points to screen coordinates
        Assign empty array to points
        For i from 0 to body.trail.length - 1
            Assign body.trail[i] to p
            Push this.coordSystem.worldToScreen(p.x, p.y) to points
        
        // Choose trail color (blue for barycenters, body color for others)
        If body.isBarycenter
            Assign '#0000FF' to color
        Otherwise
            Assign body.color to color
        
        // Convert color to semi-transparent rgba
        Assign colorToRgba(color, 0.4) to trailColor
        
        // Draw the trail
        Assign trailColor to this.ctx.strokeStyle
        Assign 1 to this.ctx.lineWidth
        Call this.ctx.beginPath()
        Call this.ctx.moveTo(points[0].x, points[0].y)
        
        For i from 1 to points.length - 1
            Call this.ctx.lineTo(points[i].x, points[i].y)
        
        Call this.ctx.stroke()

    // MARK: GRID RENDERING
    drawGrid()
        // Grid settings
        Assign 100 to spacing
        Assign 'rgba(255,255,255,0.08)' to gridColor
        
        // Save current drawing state
        Call this.ctx.save()
        
        // Set up dotted line style
        Assign gridColor to this.ctx.strokeStyle
        Assign 1 to this.ctx.lineWidth
        Call this.ctx.setLineDash([2, 6])
        
        // Draw vertical lines
        For x from 0 to this.canvas.width step spacing
            Call this.ctx.beginPath()
            Call this.ctx.moveTo(x + 0.5, 0)
            Call this.ctx.lineTo(x + 0.5, this.canvas.height)
            Call this.ctx.stroke()
        
        // Draw horizontal lines
        For y from 0 to this.canvas.height step spacing
            Call this.ctx.beginPath()
            Call this.ctx.moveTo(0, y + 0.5)
            Call this.ctx.lineTo(this.canvas.width, y + 0.5)
            Call this.ctx.stroke()
        
        // Restore previous drawing state
        Call this.ctx.restore()

    // MARK: UI RENDERING
    drawUI(cameraMode, timeMultiplier, simulationTime, targetPlanet, useRealScale, planetScaleBoost)
        // Set up text style
        Assign 'white' to this.ctx.fillStyle
        Assign '13px Arial' to this.ctx.font
        Assign 'left' to this.ctx.textAlign
        
        // Starting position for UI text
        Assign 10 to x
        Assign 20 to y
        
        // Calculate time display
        Assign Math.round(simulationTime divided by 86400) to days
        Assign days divided by 365.25 to years
        
        If years is greater than or equal to 1
            Assign years.toFixed(2) concatenated with ' years' to timeText
        Otherwise
            Assign days concatenated with ' days' to timeText
        
        // Draw speed and time
        Call this.ctx.fillText('Speed: ' concatenated with Math.round(timeMultiplier) concatenated with 'x', x, y)
        Add 18 to y
        Call this.ctx.fillText('Elapsed: ' concatenated with timeText, x, y)
        Add 22 to y
        
        // If a planet is selected, show information
        If targetPlanet exists
            // Define fun facts for each planet
            Define facts as object:
                'Sun': {line: 'A huge, hot star that lights our solar system.', size: 'Huge'}
                'Mercury': {line: 'Tiny and speedy, very close to the Sun.', size: 'Tiny'}
                'Venus': {line: 'Cloudy and bright with a very hot surface.', size: 'Small'}
                'Earth': {line: 'Our home — the only planet known with life.', size: 'Medium'}
                'Moon': {line: 'Earth's companion that lights the night.', size: 'Tiny'}
                'Mars': {line: 'The red planet with big volcanoes and valleys.', size: 'Small'}
                'Jupiter': {line: 'A giant gas planet with a huge storm.', size: 'Huge'}
                'Saturn': {line: 'Famous for its beautiful rings.', size: 'Huge'}
                'Uranus': {line: 'A tilted, icy world that looks blue-green.', size: 'Large'}
                'Neptune': {line: 'A windy, distant blue planet.', size: 'Large'}
            
            // Get info for selected planet (or default)
            If facts[targetPlanet] exists
                Assign facts[targetPlanet] to info
            Otherwise
                Assign {line: 'A mysterious world to discover!', size: 'Unknown'} to info
            
            // Draw planet name in yellow
            Assign '#ffeb3b' to this.ctx.fillStyle
            Call this.ctx.fillText(targetPlanet, x, y)
            Add 18 to y
            
            // Draw fact and size in white
            Assign 'white' to this.ctx.fillStyle
            Call this.ctx.fillText(info.line, x, y)
            Add 18 to y
            Call this.ctx.fillText('Size: ' concatenated with info.size, x, y)
            Add 18 to y
            
            // Draw hint in lighter white
            Assign 'rgba(255,255,255,0.85)' to this.ctx.fillStyle
        
        Otherwise
            // No planet selected, show instructions
            Call this.ctx.fillText('Pick a planet to see a fun fact!', x, y)
            Add 18 to y
            Call this.ctx.fillText('Modes: 1 Solar System • 2 Inner Planets • 3 Planet Focus', x, y)
            Add 18 to y
            Call this.ctx.fillText('Current view: ' concatenated with cameraMode, x, y)

End Define class Renderer


// MARK: HELPER FUNCTIONS

Define function colorToRgba(color, alpha)
    // Convert hex color to rgba string
    If color is a string AND color starts with '#' AND color.length equals 7
        // Extract RGB values from hex
        Assign parseInt(color.slice(1,3), 16) to r
        Assign parseInt(color.slice(3,5), 16) to g
        Assign parseInt(color.slice(5,7), 16) to b
        
        // Return rgba string
        Return 'rgba(' concatenated with r concatenated with ', ' concatenated with g concatenated with ', ' concatenated with b concatenated with ', ' concatenated with alpha concatenated with ')'
    
    // If not valid hex, return original color
    Return color

End Define function colorToRgba
