    isBarycenter(bodyKey)
        If Body.planetData does not have key bodyKey
            Return false
        If Body.planetData[bodyKey] has field isBarycenter and Body.planetData[bodyKey].isBarycenter is equal to true
            Return true
        Otherwise
            Return false

    getGravitationalParameter(bodyKey)
        Assign Body.planetData[bodyKey] to bodyData

        If bodyData exists AND bodyData has field orbit AND bodyData.orbit has field gravitationalParameter
            Return bodyData.orbit.gravitationalParameter

        If bodyData exists AND bodyData has field orbit AND bodyData.orbit has field centralBody
            Assign bodyData.orbit.centralBody to centralBody
        Otherwise
            Assign null to centralBody

        If centralBody is equal to 'sun'
            Return Body.GM_SUN
        Otherwise if centralBody is equal to 'earthMoonBarycenter'
            Return Body.GM_EARTH_MOON

        Return Body.GM_SUN

    getDisplayRadius()
        Assign this.mass divided by 1e20 to tmp
        Assign log10(tmp) to val
        Assign val * 2 to scaled
        If scaled is less than 2
            Return 2
        Otherwise
            Return scaled

    getWorldRadius()
        Return this.radius * 1000

End Define class Body
